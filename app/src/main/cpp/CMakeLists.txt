cmake_minimum_required(VERSION 3.22.1)
project(repasumi_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Pinned source at vendor/llama.cpp, checked out by CI (not a remote runtime dependency).
set(LLAMA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/llama.cpp")
if(NOT EXISTS "${LLAMA_ROOT}/include/llama.h")
    message(FATAL_ERROR "Falta vendor/llama.cpp. Consulta README para descargar v0.9.0.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/llama-build")
add_library(repasumi_native SHARED local_llm.cpp)
target_include_directories(repasumi_native PRIVATE "${LLAMA_ROOT}/include" "${LLAMA_ROOT}/ggml/include")
target_link_libraries(repasumi_native PRIVATE llama log android)
